cmake_minimum_required(VERSION 3.22)
project(phonebot LANGUAGES CXX)
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libnode.so)
add_library(phonebot SHARED engine.cpp)
target_include_directories(phonebot SYSTEM PRIVATE include)
target_compile_features(phonebot PRIVATE cxx_std_20)
target_compile_options(phonebot PRIVATE -Wall -Wextra -Werror)
target_link_libraries(phonebot node)
